# The Feast of the Fez — who receives the fez at the party.
# One shared rule, so the activity, its locked invite rule and its payout all agree.
# Honorees = the host's sons AND wards: male, Shriner, aged 12-15, not yet initiated.
# Hostages (warden_hostage) are a different relation and are deliberately NOT included.
# $HOST$ = the Shriner father/guardian presenting the fez.
ate_fez_eligible_son_trigger = {
	is_male = yes
	age >= 12
	age < 16
	faith = faith:shriner
	NOT = { has_character_modifier = ate_shriner_initiated_modifier }
	OR = {
		is_child_of = $HOST$
		any_relation = {
			type = guardian
			this = $HOST$
		}
	}
}

# Is the Feast of the Fez being held at this scale? $SCALE$ = modest / grand / extravagant.
# Needs scope:activity (available in on_complete and everything it calls).
ate_fez_scale_trigger = {
	scope:activity = {
		has_activity_option = {
			category = ate_fez_party_scale
			option = ate_fez_scale_$SCALE$
		}
	}
}

			variable = ate_fez_honorees

		}
	}
}

# #16 Adult Too: an attending son of the host under 12, bossy/ambitious/stubborn, not already an honoree.

# Is this character on the honoree list? Scope: character.
ate_fez_is_honoree_trigger = {
	scope:activity = {
		is_target_in_variable_list = {
			name = ate_fez_honorees

		}
	}
}

# #2 The Elder Brother: an attending son of the host who got his fez BEFORE this party. Scope: character.
ate_fez_elder_brother_candidate_trigger = {
	is_male = yes
	age >= 12
	is_child_of = scope:host
	faith = faith:shriner
	NOT = { ate_fez_is_honoree_trigger = yes }
	OR = {
		has_character_flag = ate_fez_received
		has_character_modifier = ate_shriner_initiated_modifier
		is_adult = yes
	}
}

# #3 The Potentate's Envoy: the Shriner head of faith exists and is not the host. Scope: any.
ate_fez_has_potentate_trigger = {
	exists = faith:shriner.religious_head
	faith:shriner.religious_head = {
		is_alive = yes
		NOT = { this = scope:host }
	}
}

# #4 A Knight of the Fez of Islam: the holy order exists and its leader is not the host. Scope: any.
ate_fez_has_fez_knight_trigger = {
	exists = global_var:ate_fez_of_islam_title
	exists = global_var:ate_fez_of_islam_title.holder
	global_var:ate_fez_of_islam_title.holder = {
		is_alive = yes
		NOT = { this = scope:host }
	}
}

# #5 / #14 — betrothals. The young one to be matched ($PARTNER$) must be free; the guest of rank must have a
# free child who could marry them (vanilla could_marry_character_trigger: sex, faith, consanguinity).
ate_fez_free_to_match_trigger = {
	is_alive = yes
	is_married = no
	is_betrothed = no
	is_concubine = no
}

# Scope: a guest's child. $PARTNER$ = the host's boy/daughter.
ate_fez_match_candidate_trigger = {
	ate_fez_free_to_match_trigger = yes
	age >= 6
	age <= 20
	NOT = { is_close_family_of = $PARTNER$ }
	could_marry_character_trigger = { CHARACTER = $PARTNER$ }
}

# Scope: an attending guest. A landed, AI guest (count or higher) with a child for $PARTNER$.
ate_fez_suitor_family_trigger = {
	NOT = { this = scope:host }
	is_ai = yes
	is_adult = yes
	highest_held_title_tier >= tier_county
	any_child = { ate_fez_match_candidate_trigger = { PARTNER = $PARTNER$ } }
}

# #5 A Match Proposed: an attending honoree free to marry. Scope: character.
ate_fez_match_boy_trigger = {
	ate_fez_attending_honoree_trigger = yes
	ate_fez_free_to_match_trigger = yes
}

# #5: is there an attending honoree AND a guest with a match for him? Scope: the host.
ate_fez_has_boy_match_trigger = {
	scope:activity = {
		any_in_list = {
			variable = ate_fez_honorees
			ate_fez_match_boy_trigger = yes
			save_temporary_scope_as = ate_fez_match_check
			scope:activity = {
				any_attending_character = { ate_fez_suitor_family_trigger = { PARTNER = scope:ate_fez_match_check } }
			}
		}
	}
}

# #14 A Suitor's Eye: an attending under-age daughter free to marry AND a guest with a match for her. Scope: the host.
ate_fez_has_daughter_match_trigger = {
	scope:activity = {

			ate_fez_daughter_candidate_trigger = yes
			ate_fez_free_to_match_trigger = yes
			save_temporary_scope_as = ate_fez_match_check
			scope:activity = {
				any_attending_character = { ate_fez_suitor_family_trigger = { PARTNER = scope:ate_fez_match_check } }
			}
		}
	}
}

			is_alive = yes
			NOT = { this = scope:host }
		}
	}
}

		has_trait = hashishiyah
		has_trait = comfort_eater
		has_trait = gluttonous
		has_trait = profligate
		has_trait = greedy
		has_trait = lifestyle_reveler
	}

# #17 d: can the boy catch scope:ate_fez_vice? Not already his, no opposite trait, a free personality slot for
# Greedy. A Gluttonous father passes on Comfort Eater. Scope: the boy.
ate_fez_boy_can_catch_vice_trigger = {
	OR = {
		AND = {
			scope:ate_fez_vice = flag:drunkard
			NOT = { has_trait = drunkard }
		}
		AND = {
			scope:ate_fez_vice = flag:hashishiyah
			NOT = { has_trait = hashishiyah }
		}
		AND = {
			OR = {
				scope:ate_fez_vice = flag:comfort_eater
				scope:ate_fez_vice = flag:gluttonous
			}
			NOT = { has_trait = comfort_eater }
			NOT = { has_trait = inappetetic }
		}
		AND = {
			scope:ate_fez_vice = flag:profligate
			NOT = { has_trait = profligate }
			NOT = { has_trait = improvident }
		}
		AND = {
			scope:ate_fez_vice = flag:greedy
			NOT = { has_trait = greedy }
			NOT = { has_trait = generous }
			number_of_personality_traits < 3
		}
		AND = {
			scope:ate_fez_vice = flag:lifestyle_reveler
			NOT = { has_trait = lifestyle_reveler }
		}
	}
}
