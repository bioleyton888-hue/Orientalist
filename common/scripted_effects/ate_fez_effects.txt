# The Feast of the Fez — effects used by activity_ate_fez_ceremony.

# Stores every eligible son and ward of the host in the activity's ate_fez_honorees list.
# Called from on_start. Scope: the activity. Needs scope:host and scope:activity.
# The list is the record of who the party was thrown for; the payout only pays those who attended.
ate_fez_collect_honorees_effect = {
	scope:host = {
		every_child = {
			limit = { ate_fez_eligible_son_trigger = { HOST = scope:host } }
			save_temporary_scope_as = ate_fez_boy
			scope:activity = {
				add_to_variable_list = { name = ate_fez_honorees target = scope:ate_fez_boy }
			}
		}
		every_relation = {
			type = ward
			limit = {
				NOT = { is_child_of = scope:host } # a son who is also a ward is already in the list
				ate_fez_eligible_son_trigger = { HOST = scope:host }
			}
			save_temporary_scope_as = ate_fez_boy
			scope:activity = {
				add_to_variable_list = { name = ate_fez_honorees target = scope:ate_fez_boy }
			}
		}
	}
}

# Host payout at the end of the party, shown in the conclusion screen.
# Per scale (plan section 4.3):
#   Modest       = old ate_shriner.0015.a: boys initiated; host +50 prestige, -10 stress
#   Grand        = old ate_shriner.0015.b: + boys' opinion of the host; host +150 prestige, +50 piety
#   Extravagant  = + boys get Presented at the Great Feast (20 y); host +350 prestige, dynasty prestige,
#                  Talk of the Lodge (5 y)
# Called from on_complete for the host. Needs scope:host and scope:activity.
ate_fez_host_payout_effect = {
	scope:activity = {
		add_activity_log_entry = {
			key = ate_fez_honorees_log
			tags = { completed }
			show_in_conclusion = yes
			character = scope:host
			score = 100

			scope:activity = {
				every_in_list = {
					variable = ate_fez_honorees
					limit = {
						is_alive = yes
						is_participant_in_activity = scope:activity
						NOT = { has_character_modifier = ate_shriner_initiated_modifier }
					}
					custom = ate_fez_every_honoree_tt
					add_character_modifier = {
						modifier = ate_shriner_initiated_modifier
						years = 6
					}
					if = {
						limit = { NOT = { ate_fez_scale_trigger = { SCALE = modest } } }
						add_opinion = {
							modifier = ate_fez_presented_by_father_opinion
							target = scope:host
						}
					}
					if = {
						limit = { ate_fez_scale_trigger = { SCALE = extravagant } }
						add_character_modifier = {
							modifier = ate_fez_presented_great_feast_modifier
							years = 20
						}
					}
				}
			}
			scope:host = {
				if = {
					limit = { ate_fez_scale_trigger = { SCALE = extravagant } }
					add_prestige = major_prestige_gain
					dynasty ?= { add_dynasty_prestige = minor_dynasty_prestige_gain }
					add_character_modifier = {
						modifier = ate_fez_talk_of_the_lodge_modifier
						years = 5
					}
				}
				else_if = {
					limit = { ate_fez_scale_trigger = { SCALE = grand } }
					add_prestige = medium_prestige_gain
					add_piety = minor_piety_gain
				}
				else = {
					add_prestige = 50
					add_stress = -10
				}
			}
		}
	}
}

# Pays the intent the character picked. Players see it in the conclusion screen; AI just gets it.
# Called from on_complete for every participant. Needs scope:activity.
ate_fez_intent_payout_effect = {
	if = {
		limit = { is_ai = no }
		save_scope_as = ate_fez_intent_owner
		scope:activity = {
			add_activity_log_entry = {
				key = ate_fez_intent_reward_log
				tags = { completed }
				show_in_conclusion = yes
				character = scope:ate_fez_intent_owner
				score = 50

				scope:ate_fez_intent_owner = { ate_fez_intent_reward_effect = yes }
			}
		}
	}
	else = {
		ate_fez_intent_reward_effect = yes
	}
}

# The rewards themselves, growing with the scale: Modest = minor, Grand = medium, Extravagant = major.
# Needs scope:activity.
ate_fez_intent_reward_effect = {
	if = {
		limit = { has_activity_intent = ate_fez_party_intent }
		if = {
			limit = { ate_fez_scale_trigger = { SCALE = extravagant } }
			add_stress = major_stress_loss
		}
		else_if = {
			limit = { ate_fez_scale_trigger = { SCALE = grand } }
			add_stress = medium_stress_loss
		}
		else = {
			add_stress = minor_stress_loss
		}
	}
	else_if = {
		limit = { has_activity_intent = ate_fez_important_intent }
		if = {
			limit = { ate_fez_scale_trigger = { SCALE = extravagant } }
			add_piety = major_piety_gain
		}
		else_if = {
			limit = { ate_fez_scale_trigger = { SCALE = grand } }
			add_piety = medium_piety_gain
		}
		else = {
			add_piety = minor_piety_gain
		}
	}
	else_if = {
		limit = { has_activity_intent = ate_fez_look_intent }
		if = {
			limit = { ate_fez_scale_trigger = { SCALE = extravagant } }
			add_prestige = major_prestige_gain
		}
		else_if = {
			limit = { ate_fez_scale_trigger = { SCALE = grand } }
			add_prestige = medium_prestige_gain
		}
		else = {
			add_prestige = minor_prestige_gain
		}
	}
}

# Saves up to five ATTENDING boys as scope:ate_fez_boy_1..5 (for event portraits and names), and
# scope:ate_fez_many_boys when more than five are present. Used by the phase events.
# Needs scope:activity. Each pick is guarded so an unset previous scope is never compared.
ate_fez_save_present_boys_effect = {
	scope:activity = {
		every_in_list = {
			variable = ate_fez_honorees
			limit = {
				is_alive = yes
				is_participant_in_activity = scope:activity
			}
			add_to_temporary_list = ate_fez_present_boys
		}
	}
	random_in_list = {
		list = ate_fez_present_boys
		save_scope_as = ate_fez_boy_1
	}
	if = {
		limit = { exists = scope:ate_fez_boy_1 }
		random_in_list = {
			list = ate_fez_present_boys
			limit = { NOT = { this = scope:ate_fez_boy_1 } }
			save_scope_as = ate_fez_boy_2
		}
	}
	if = {
		limit = { exists = scope:ate_fez_boy_2 }
		random_in_list = {
			list = ate_fez_present_boys
			limit = {
				NOR = {
					this = scope:ate_fez_boy_1
					this = scope:ate_fez_boy_2
				}
			}
			save_scope_as = ate_fez_boy_3
		}
	}
	if = {
		limit = { exists = scope:ate_fez_boy_3 }
		random_in_list = {
			list = ate_fez_present_boys
			limit = {
				NOR = {
					this = scope:ate_fez_boy_1
					this = scope:ate_fez_boy_2
					this = scope:ate_fez_boy_3
				}
			}
			save_scope_as = ate_fez_boy_4
		}
	}
	if = {
		limit = { exists = scope:ate_fez_boy_4 }
		random_in_list = {
			list = ate_fez_present_boys
			limit = {
				NOR = {
					this = scope:ate_fez_boy_1
					this = scope:ate_fez_boy_2
					this = scope:ate_fez_boy_3
					this = scope:ate_fez_boy_4
				}
			}
			save_scope_as = ate_fez_boy_5
		}
	}
	if = {
		limit = {
			any_in_list = {
				list = ate_fez_present_boys
				count > 5
			}
		}
		save_scope_value_as = { name = ate_fez_many_boys value = yes }
	}
}
