# Fired on_phase_active of ate_fez_phase_procession, host only.
########################################

	type = activity_event
	title = ate_fez_ceremony.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { ate_fez_scale_trigger = { SCALE = extravagant } }
				desc = ate_fez_ceremony.0001.desc.extravagant

			triggered_desc = {
				trigger = { ate_fez_scale_trigger = { SCALE = grand } }
				desc = ate_fez_ceremony.0001.desc.grand

			desc = ate_fez_ceremony.0001.desc.modest
		}
		triggered_desc = {
			trigger = {
				ate_fez_scale_trigger = { SCALE = extravagant }

	override_background = { reference = ate_fez_lodge_hall }

		character = root
		animation = personality_honorable

		exists = scope:activity

		name = ate_fez_ceremony.0001.a
	}

########################################
# ate_fez_ceremony.0002 — The Presentation
# Fired on_phase_active of ate_fez_phase_presentation, host only.
# ONE event for every boy: host left, up to five boys in the other portrait slots.
########################################
ate_fez_ceremony.0002 = {
	type = activity_event
	title = ate_fez_ceremony.0002.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { NOT = { exists = scope:ate_fez_boy_1 } }
				desc = ate_fez_ceremony.0002.desc.none
			}
			triggered_desc = {
				trigger = { NOT = { exists = scope:ate_fez_boy_2 } }
				desc = ate_fez_ceremony.0002.desc.one
			}
			triggered_desc = {
				trigger = { exists = scope:ate_fez_many_boys }
				desc = ate_fez_ceremony.0002.desc.many
			}
			desc = ate_fez_ceremony.0002.desc.several
		}
	}
	theme = feast_activity
	override_background = { reference = ate_fez_lodge_hall }

	left_portrait = {
		character = root
		animation = personality_honorable
	}
	right_portrait = {
		character = scope:ate_fez_boy_1
		animation = happiness
	}
	center_portrait = {
		character = scope:ate_fez_boy_2
		animation = happiness
	}
	lower_left_portrait = scope:ate_fez_boy_3
	lower_center_portrait = scope:ate_fez_boy_4
	lower_right_portrait = scope:ate_fez_boy_5

	trigger = {
		exists = scope:activity
	}

	immediate = {
		ate_fez_save_present_boys_effect = yes
		add_character_flag = {
			flag = ate_fez_busy_in_event
			days = 2
		}
		# The fez is placed on their heads: from now on their portraits show it (children need this flag)
		scope:activity = {
			every_in_list = {
				variable = ate_fez_honorees
				limit = { ate_fez_attending_honoree_trigger = yes }
				add_character_flag = ate_fez_received
			}
		}
	}

	# Solemn — the oaths matter more than the feast
	option = {
		name = ate_fez_ceremony.0002.a
		add_piety = 25
		stress_impact = {
			zealous = minor_stress_impact_loss
			cynical = minor_stress_impact_gain
		}
		ai_chance = {
			base = 10
		}
	}

	# Proud — let the whole hall look at them
	option = {
		name = ate_fez_ceremony.0002.b
		add_prestige = 25
		stress_impact = {
			arrogant = minor_stress_impact_loss
			humble = minor_stress_impact_gain
		}
		ai_chance = {
			base = 10
		}
	}

	# Joking — the fez fits nobody at first
	option = {
		name = ate_fez_ceremony.0002.c
		add_stress = minor_stress_loss
		scope:activity = {
			every_in_list = {
				variable = ate_fez_honorees
				limit = {
					is_alive = yes
					is_participant_in_activity = scope:activity
				}
				custom = ate_fez_every_present_boy_tt
				add_opinion = {
					target = scope:host
					modifier = pleased_opinion
					opinion = 10
				}
			}
		}
		stress_impact = {
			gregarious = minor_stress_impact_loss
			shy = minor_stress_impact_gain
		}
		ai_chance = {
			base = 10
		}
	}

	# 2026-10-25 (user): #1 "Zub-dah Fingers" happens during the ceremony, not the parade. The Presentation lasts
	# 3 days and has no weekly pool, so #1 is chained from here; its own trigger still filters it.
	after = {
		random = {
			chance = 100    # DEBUG: release value 30
			trigger_event = {
				id = ate_fez_ceremony.1001
				days = 1
			}
		}
	}
}

ate_fez_ceremony.1001 = {
	type = activity_event
	title = ate_fez_ceremony.1001.t
	desc = ate_fez_ceremony.1001.desc
	theme = feast_activity
	override_background = { reference = ate_fez_lodge_hall }

	left_portrait = {
		character = root
		animation = personality_honorable
	}
	right_portrait = {
		character = scope:ate_fez_target_boy
		animation = shock
	}

	trigger = {
		exists = scope:activity
		scope:activity = { NOT = { has_variable = ate_fez_done_1 } }
		ate_fez_has_attending_honoree_trigger = yes
	}

	immediate = {
		ate_fez_mark_pool_event_effect = { EVENT = 1 }
		ate_fez_pick_attending_honoree_effect = yes
	}

	# a: laugh it off (2026-10-25 user: + a step toward friendship, zealous Shriner guests frown)
	option = {
		name = ate_fez_ceremony.1001.a
		add_stress = minor_stress_loss
		send_interface_toast = {
			type = event_toast_effect_good
			title = ate_fez_ceremony.1001.a.toast
			left_icon = scope:ate_fez_target_boy
			scope:ate_fez_target_boy = {
				add_opinion = {
					target = root
					modifier = pleased_opinion
					opinion = 10
				}
			}
			progress_towards_friend_effect = {
				REASON = ate_fez_friend_laughed_off_fez
				CHARACTER = scope:ate_fez_target_boy
				OPINION = 0
			}
		}
		scope:activity = {
			every_attending_character = {
				limit = {
					NOT = { this = root }
					has_faith = faith:shriner
					has_trait = zealous
				}
				custom = ate_fez_every_zealous_shriner_guest_tt
				add_opinion = {
					target = root
					modifier = disappointed_opinion
					opinion = -5
				}
			}
		}
	}

	# b: scold him (he wears the broken fez)
	option = {
		name = ate_fez_ceremony.1001.b
		custom_tooltip = ate_fez_ceremony.1001.b.tt
		add_prestige = miniscule_prestige_value
		scope:ate_fez_target_boy = {
			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = -10
			}
		}
	}

	# c: have it re-fitted on the spot
	option = {
		name = ate_fez_ceremony.1001.c
		remove_short_term_gold = tiny_gold_value
		add_prestige = miniscule_prestige_value
	}

	# d: fix it with your own hands (2026-10-25 user: high learning or stewardship; "high" = 12, the mod's usual bar)
	option = {
		name = ate_fez_ceremony.1001.d
		trigger = {
			OR = {
				learning >= 12
				stewardship >= 12
			}
		}
		add_prestige = miniscule_prestige_value
		add_piety = miniscule_piety_value
		scope:ate_fez_target_boy = {
			add_opinion = {
				target = root
				modifier = pleased_opinion
				opinion = 5
			}
		}
	}
}
